const { onSchedule } = require('firebase-functions/v2/scheduler'); const { onCall, HttpsError } = require('firebase-functions/v2/https'); const { logger } = require('firebase-functions'); const admin = require('firebase-admin'); admin.initializeApp(); const dateKey = (d) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`; // Employee IDs aren't valid email addresses, and Firebase Auth is built // around email/password. This maps EMPEG001 -> empeg001@edubotics-attendance.internal // so we can use Firebase Auth's normal email/password mechanism while the // app UI still only ever shows "Employee ID" to the user. const SYNTHETIC_EMAIL_DOMAIN = 'edubotics-attendance.internal'; const toSyntheticEmail = (employeeId) => `${employeeId.toLowerCase()}@${SYNTHETIC_EMAIL_DOMAIN}`; // This backend job runs even when the app is closed. An office record remains // pending until an admin accepts the request and chooses the actual checkout // time. WFH records are completed automatically at the boundary. exports.closeOpenAttendanceAtMidnight = onSchedule({ schedule: '59 23 * * *', timeZone: 'Asia/Kolkata' }, async () => { const now = new Date(); const indiaNow = new Date(now.toLocaleString('en-US', { timeZone: 'Asia/Kolkata' })); const date = dateKey(indiaNow); const root = admin.database().ref(); const attendance = await root.child('Attendance').once('value'); const updates = {}; attendance.forEach((employee) => { const record = employee.child(date).val(); if (!record || record.status !== 'Checked In') return; const employeeId = employee.key; updates[`Attendance/${employeeId}/${date}/autoCheckedOutAt`] = now.toISOString(); if (record.workFromHome === true) { updates[`Attendance/${employeeId}/${date}/punchOut`] = '11:59 PM'; updates[`Attendance/${employeeId}/${date}/status`] = 'Checked Out'; updates[`Attendance/${employeeId}/${date}/attendanceStatus`] = 'Full Day (WFH auto checkout)'; } else { updates[`Attendance/${employeeId}/${date}/status`] = 'Auto Checkout Pending'; updates[`Attendance/${employeeId}/${date}/attendanceStatus`] = 'Auto punch-out pending admin approval'; updates[`PunchRequests/${employeeId}/${date}`] = { employeeId, date, type: 'auto_checkout', status: 'pending', punchIn: record.punchIn, suggestedPunchOut: '11:59 PM', message: 'Employee did not check out. Verify and select the actual checkout time.', createdAt: now.toISOString() }; } }); if (Object.keys(updates).length) await root.update(updates); logger.info(`Midnight attendance processing complete for ${date}`); }); /** * Callable function used by the Flutter app's login screen. * * The app can no longer read /users/{employeeId} directly to check a * password once the database rules require auth — this function runs * with admin privileges (bypassing all rules) so it's the one place * still allowed to see the legacy plaintext password field and decide * whether a login attempt is valid. * * On first successful login for a given employee, it also *creates* * their Firebase Auth account on the spot (using their existing RTDB * password), so there's no separate one-off migration script to run — * every employee gets migrated automatically the next time they log in. * * Returns a custom token the client signs in with via * FirebaseAuth.instance.signInWithCustomToken(token), which is what * actually satisfies "auth != null" in the database rules from then on. */ exports.login = onCall(async (request) => { const employeeIdRaw = request.data?.employeeId; const password = request.data?.password; if (typeof employeeIdRaw !== 'string' || typeof password !== 'string' || !employeeIdRaw.trim() || !password) { throw new HttpsError('invalid-argument', 'Employee ID and password are required.'); } const employeeId = employeeIdRaw.trim().toUpperCase(); const db = admin.database(); // Same lookup logic as the old client-side _findUser: try the RTDB key // directly first, then fall back to scanning for an employeeId field // (covers older records where the key doesn't match the ID). let userSnap = await db.ref(`users/${employeeId}`).once('value'); let resolvedKey = employeeId; if (!userSnap.exists()) { const allUsers = await db.ref('users').once('value'); allUsers.forEach((child) => { const val = child.val(); if (val && String(val.employeeId || '').trim().toUpperCase() === employeeId) { userSnap = child; resolvedKey = child.key; } }); } if (!userSnap.exists()) { throw new HttpsError('not-found', 'Employee ID not found.'); } const user = userSnap.val(); const storedPassword = String(user.password ?? ''); if (storedPassword !== password) { throw new HttpsError('permission-denied', 'Wrong password.'); } const role = String(user.role ?? 'employee').trim().toLowerCase(); const name = user.name ?? resolvedKey; const email = toSyntheticEmail(resolvedKey); // Ensure a Firebase Auth account exists for this employee, using their // Employee ID as the Auth UID. Using employeeId as the UID directly is // what lets the database rules use simple checks like // "auth.uid === $employeeId" for per-user data. try { await admin.auth().getUser(resolvedKey); } catch (err) { if (err.code === 'auth/user-not-found') { await admin.auth().createUser({ uid: resolvedKey, email, password, displayName: name, }); logger.info(`Migrated ${resolvedKey} to Firebase Auth on first login.`); } else { throw err; } } // Custom claims let the database rules check role without another // database read: auth.token.role === 'admin', etc. await admin.auth().setCustomUserClaims(resolvedKey, { role }); const customToken = await admin.auth().createCustomToken(resolvedKey, { role }); return { customToken, employeeId: resolvedKey, name, role, registeredDeviceId: user.registeredDeviceId ?? null, }; });
